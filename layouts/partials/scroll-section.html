{{/* Wrapper partial for homepage sections Accepts a dict with: - num: section number (1-9) - id: semantic section slug used as the DOM id and URL hash (e.g. "privacy") - partial: name of the content partial to include (e.g., "home-hero.html") - context: the original page context for passing to content partials - tone: band tone (paper | surface | charcoal, default paper) paper => bg-background surface => bg-card border-y border-border (hairlines separate it from paper) charcoal => bg-foreground text-background All sections flow naturally with the page; the homepage is a normal scrolling reading page with no snap or exact-fit slides. Visible section number: the hero has no kicker, so content partials read `page.Store.Get "sectionNum"` (num minus 1) and pass it to section-kicker.html. */}} {{ $num := .num }} {{ $id := .id }} {{ $partial := .partial }} {{ $context := .context }} {{/* Map the tone to the full-bleed band classes */}} {{ $tone := .tone | default "paper" }} {{ $bgClass := "bg-background" }} {{ if eq $tone "surface" }} {{ $bgClass = "bg-card border-y border-border" }} {{ else if eq $tone "charcoal" }} {{ $bgClass = "bg-foreground text-background" }} {{ end }} {{ page.Store.Set "sectionNum" (sub $num 1) }}
{{ partial $partial $context }}
